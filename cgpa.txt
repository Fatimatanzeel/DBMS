-- Task: Create Table
-- -----------------------------------------------------------------------------
CREATE TABLE Student (
    Student_ID   NUMBER PRIMARY KEY,
    Student_Name VARCHAR2(50) NOT NULL,
    Gender       CHAR(1) CHECK (Gender IN ('M', 'F', 'O')),
    Department   VARCHAR2(20),
    Semester     NUMBER(2) CHECK (Semester > 0),
    Age          NUMBER(3) CHECK (Age > 0),
    CGPA         NUMBER(3, 2) CHECK (CGPA BETWEEN 0.00 AND 10.00)
);

-- -----------------------------------------------------------------------------
-- 3. Display the names and CGPAs of all students.
-- -----------------------------------------------------------------------------
SELECT Student_Name, CGPA 
FROM Student;

-- -----------------------------------------------------------------------------
-- 4. Retrieve the details of students whose CGPA is 8.0 or above.
-- -----------------------------------------------------------------------------
SELECT * 
FROM Student 
WHERE CGPA >= 8.0;

-- -----------------------------------------------------------------------------
-- 5. Retrieve students belonging to either the CSE or ECE department.
-- -----------------------------------------------------------------------------
SELECT * 
FROM Student 
WHERE Department IN ('CSE', 'ECE');

-- -----------------------------------------------------------------------------
-- 6. Find students currently studying in Semester 4 or Semester 6.
-- -----------------------------------------------------------------------------
SELECT * 
FROM Student 
WHERE Semester IN (4, 6);

-- -----------------------------------------------------------------------------
-- 7. Display students whose names start with the letter 'A'.
-- -----------------------------------------------------------------------------
SELECT * 
FROM Student 
WHERE Student_Name LIKE 'A%';

-- -----------------------------------------------------------------------------
-- 8. Display students in ascending order of their names.
-- -----------------------------------------------------------------------------
SELECT * 
FROM Student 
ORDER BY Student_Name ASC;

-- -----------------------------------------------------------------------------
-- 9. Display the three students having the highest CGPA.
-- (Oracle 12c+ FETCH clause)
-- -----------------------------------------------------------------------------
SELECT * 
FROM Student 
ORDER BY CGPA DESC 
FETCH FIRST 3 ROWS ONLY;

-- -----------------------------------------------------------------------------
-- 10. Find the average CGPA of all students.
-- -----------------------------------------------------------------------------
SELECT AVG(CGPA) AS Average_CGPA 
FROM Student;

-- -----------------------------------------------------------------------------
-- 11. Find the highest and lowest CGPA in the class.
-- -----------------------------------------------------------------------------
SELECT MAX(CGPA) AS Highest_CGPA, 
       MIN(CGPA) AS Lowest_CGPA 
FROM Student;

-- -----------------------------------------------------------------------------
-- 12. Count the number of students in each department.
-- -----------------------------------------------------------------------------
SELECT Department, COUNT(*) AS Total_Students 
FROM Student 
GROUP BY Department;
