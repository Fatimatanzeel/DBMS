-- Task 1: Create the Product table using suitable data types and constraints.
-- Fields: Product_ID, Product_Name, Category, Brand, Price, Quantity, Supplier_City
-- -----------------------------------------------------------------------------
CREATE TABLE Product (
    Product_ID    NUMBER PRIMARY KEY,
    Product_Name  VARCHAR2(100) NOT NULL,
    Category      VARCHAR2(50),
    Brand         VARCHAR2(50),
    Price         NUMBER(10, 2) CHECK (Price >= 0),
    Quantity      NUMBER(5) CHECK (Quantity >= 0),
    Supplier_City VARCHAR2(50)
);

-- -----------------------------------------------------------------------------
-- Task 3: Display all products available in the table.
-- -----------------------------------------------------------------------------
SELECT * FROM Product;

-- -----------------------------------------------------------------------------
-- Task 4: Display only Product_Name, Category, and Price.
-- -----------------------------------------------------------------------------
SELECT Product_Name, Category, Price 
FROM Product;

-- -----------------------------------------------------------------------------
-- Task 5: Find all products having a price greater than 1000.
-- -----------------------------------------------------------------------------
SELECT * 
FROM Product 
WHERE Price > 1000;

-- -----------------------------------------------------------------------------
-- Task 6: Find products whose price lies between 500 and 5000.
-- -----------------------------------------------------------------------------
SELECT * 
FROM Product 
WHERE Price BETWEEN 500 AND 5000;

-- -----------------------------------------------------------------------------
-- Task 7: Display products belonging to a particular category (e.g., 'Electronics').
-- -----------------------------------------------------------------------------
SELECT * 
FROM Product 
WHERE Category = 'Electronics';

-- -----------------------------------------------------------------------------
-- Task 8: Display products belonging to either Electronics, Stationery, or Accessories using IN.
-- -----------------------------------------------------------------------------
SELECT * 
FROM Product 
WHERE Category IN ('Electronics', 'Stationery', 'Accessories');

-- -----------------------------------------------------------------------------
-- Task 9: Find products whose names start with the letter "S".
-- -----------------------------------------------------------------------------
SELECT * 
FROM Product 
WHERE Product_Name LIKE 'S%';

-- -----------------------------------------------------------------------------
-- Task 10: Find products whose names contain the word "Book".
-- -----------------------------------------------------------------------------
SELECT * 
FROM Product 
WHERE Product_Name LIKE '%Book%';

-- -----------------------------------------------------------------------------
-- Task 11: Display all products in ascending order of price.
-- -----------------------------------------------------------------------------
SELECT * 
FROM Product 
ORDER BY Price ASC;

-- -----------------------------------------------------------------------------
-- Task 12: Display all products in descending order of quantity.
-- -----------------------------------------------------------------------------
SELECT * 
FROM Product 
ORDER BY Quantity DESC;

-- -----------------------------------------------------------------------------
-- Task 13: Find the most expensive and least expensive product using MAX() and MIN().
-- -----------------------------------------------------------------------------
SELECT MAX(Price) AS Most_Expensive, 
       MIN(Price) AS Least_Expensive 
FROM Product;

-- -----------------------------------------------------------------------------
-- Task 14: Calculate the average price of all products using AVG().
-- -----------------------------------------------------------------------------
SELECT AVG(Price) AS Average_Price 
FROM Product;

-- -----------------------------------------------------------------------------
-- Task 15: Calculate the total quantity of products available using SUM().
-- -----------------------------------------------------------------------------
SELECT SUM(Quantity) AS Total_Quantity 
FROM Product;

-- -----------------------------------------------------------------------------
-- Task 16: Count the total number of products using COUNT().
-- -----------------------------------------------------------------------------
SELECT COUNT(*) AS Total_Products 
FROM Product;

-- -----------------------------------------------------------------------------
-- Task 17: Display the different product categories using DISTINCT.
-- -----------------------------------------------------------------------------
SELECT DISTINCT Category 
FROM Product;

-- -----------------------------------------------------------------------------
-- Task 18: Count the number of products available in each category using GROUP BY.
-- -----------------------------------------------------------------------------
SELECT Category, COUNT(*) AS Product_Count 
FROM Product 
GROUP BY Category;

-- -----------------------------------------------------------------------------
-- Task 19: Display only those categories having more than three products using HAVING.
-- -----------------------------------------------------------------------------
SELECT Category, COUNT(*) AS Product_Count 
FROM Product 
GROUP BY Category 
HAVING COUNT(*) > 3;

-- -----------------------------------------------------------------------------
-- Task 20: Increase the price of products belonging to a particular category by 10%.
-- -----------------------------------------------------------------------------
UPDATE Product 
SET Price = Price * 1.10 
WHERE Category = 'Electronics';

-- -----------------------------------------------------------------------------
-- Task 21: Update the quantity of a selected product.

-- -----------------------------------------------------------------------------
-- Task 22: Delete products whose quantity is zero.
-- -----------------------------------------------------------------------------
DELETE FROM Product 
WHERE Quantity = 0;
COMMIT;
